<?php

namespace Tests\Unit\Http\Requests\Master\Doctor;

use App\Http\Requests\Master\Doctor\SaveRequest;
use Illuminate\Support\Facades\Validator;
use PHPUnit\Framework\Attributes\DataProvider;
use Tests\TestCase;

class SaveRequestTest extends TestCase
{
    private function rules(): array
    {
        return (new SaveRequest())->rules();
    }

    public static function passesDataProvider(): array
    {
        return [
            '正常なデータ（type=1）' => [
                ['doctor_code' => 'D001', 'name' => '山田太郎', 'type' => 1, 'memo' => '内科'],
            ],
            '正常なデータ（type=2）' => [
                ['doctor_code' => 'D002', 'name' => '鈴木花子', 'type' => 2, 'memo' => '外科'],
            ],
            'memoはnullを許容する' => [
                ['doctor_code' => 'D001', 'name' => '山田太郎', 'type' => 1, 'memo' => null],
            ],
            'doctor_codeが255文字ちょうど' => [
                ['doctor_code' => str_repeat('a', 255), 'name' => '山田太郎', 'type' => 1, 'memo' => null],
            ],
            'nameが255文字ちょうど' => [
                ['doctor_code' => 'D001', 'name' => str_repeat('あ', 255), 'type' => 1, 'memo' => null],
            ],
        ];
    }

    #[DataProvider('passesDataProvider')]
    public function test_バリデーションを通過すること(array $data): void
    {
        $validator = Validator::make($data, $this->rules());

        $this->assertTrue($validator->passes());
    }

    public static function failsDataProvider(): array
    {
        return [
            'doctor_codeが存在しない' => [
                ['name' => '山田太郎', 'type' => 1, 'memo' => null],
                'doctor_code',
            ],
            'doctor_codeが空' => [
                ['doctor_code' => '', 'name' => '山田太郎', 'type' => 1, 'memo' => null],
                'doctor_code',
            ],
            'doctor_codeが256文字以上' => [
                ['doctor_code' => str_repeat('a', 256), 'name' => '山田太郎', 'type' => 1, 'memo' => null],
                'doctor_code',
            ],
            'nameが存在しない' => [
                ['doctor_code' => 'D001', 'type' => 1, 'memo' => null],
                'name',
            ],
            'nameが空' => [
                ['doctor_code' => 'D001', 'name' => '', 'type' => 1, 'memo' => null],
                'name',
            ],
            'nameが256文字以上' => [
                ['doctor_code' => 'D001', 'name' => str_repeat('あ', 256), 'type' => 1, 'memo' => null],
                'name',
            ],
            'typeが存在しない' => [
                ['doctor_code' => 'D001', 'name' => '山田太郎', 'memo' => null],
                'type',
            ],
            'typeがEnumに存在しない値' => [
                ['doctor_code' => 'D001', 'name' => '山田太郎', 'type' => 3, 'memo' => null],
                'type',
            ],
            'typeが文字列' => [
                ['doctor_code' => 'D001', 'name' => '山田太郎', 'type' => 'abc', 'memo' => null],
                'type',
            ],
        ];
    }

    #[DataProvider('failsDataProvider')]
    public function test_バリデーションエラーになること(array $data, string $expected_error_key): void
    {
        $validator = Validator::make($data, $this->rules());

        $this->assertTrue($validator->fails());
        $this->assertArrayHasKey($expected_error_key, $validator->errors()->toArray());
    }
}
