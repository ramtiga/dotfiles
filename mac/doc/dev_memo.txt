<?php

namespace Tests\Unit\Http\Requests\Master\Department;

use App\Http\Requests\Master\Department\SaveRequest;
use Illuminate\Support\Facades\Validator;
use PHPUnit\Framework\Attributes\DataProvider;
use Tests\TestCase;

class SaveRequestTest extends TestCase
{
    private function rules(): array
    {
        return (new SaveRequest())->rules();
    }

    public static function passesDataProvider(): array
    {
        return [
            '正常なデータ（複数行、idあり/なし混在）' => [
                [
                    'departments' => [
                        ['id' => null, 'name' => '経理部', 'memo' => '本社', 'sort_order' => 0],
                        ['id' => 1, 'name' => '総務部', 'memo' => null, 'sort_order' => 1],
                    ],
                ],
            ],
            'idはnullを許容する' => [
                [
                    'departments' => [
                        ['id' => null, 'name' => '経理部', 'memo' => null, 'sort_order' => 0],
                    ],
                ],
            ],
            'nameが255文字ちょうど' => [
                [
                    'departments' => [
                        ['id' => null, 'name' => str_repeat('あ', 255), 'memo' => null, 'sort_order' => 0],
                    ],
                ],
            ],
            'memoはnullを許容する' => [
                [
                    'departments' => [
                        ['id' => null, 'name' => '経理部', 'memo' => null, 'sort_order' => 0],
                    ],
                ],
            ],
            'memoが1000文字ちょうど' => [
                [
                    'departments' => [
                        ['id' => null, 'name' => '経理部', 'memo' => str_repeat('あ', 1000), 'sort_order' => 0],
                    ],
                ],
            ],
        ];
    }

    #[DataProvider('passesDataProvider')]
    public function test_バリデーションを通過すること(array $data): void
    {
        $validator = Validator::make($data, $this->rules());

        $this->assertTrue($validator->passes());
    }

    public static function failsDataProvider(): array
    {
        return [
            'departmentsが存在しない' => [
                [],
                'departments',
            ],
            'departmentsが空配列' => [
                ['departments' => []],
                'departments',
            ],
            'departmentsが配列でない' => [
                ['departments' => '経理部'],
                'departments',
            ],
            'idが整数でない' => [
                [
                    'departments' => [
                        ['id' => 'abc', 'name' => '経理部', 'memo' => null, 'sort_order' => 0],
                    ],
                ],
                'departments.0.id',
            ],
            'nameが空' => [
                [
                    'departments' => [
                        ['id' => null, 'name' => '', 'memo' => null, 'sort_order' => 0],
                    ],
                ],
                'departments.0.name',
            ],
            'nameが256文字以上' => [
                [
                    'departments' => [
                        ['id' => null, 'name' => str_repeat('あ', 256), 'memo' => null, 'sort_order' => 0],
                    ],
                ],
                'departments.0.name',
            ],
            'memoが1001文字以上' => [
                [
                    'departments' => [
                        ['id' => null, 'name' => '経理部', 'memo' => str_repeat('あ', 1001), 'sort_order' => 0],
                    ],
                ],
                'departments.0.memo',
            ],
            'sort_orderが存在しない' => [
                [
                    'departments' => [
                        ['id' => null, 'name' => '経理部', 'memo' => null],
                    ],
                ],
                'departments.0.sort_order',
            ],
            'sort_orderが整数でない' => [
                [
                    'departments' => [
                        ['id' => null, 'name' => '経理部', 'memo' => null, 'sort_order' => 'abc'],
                    ],
                ],
                'departments.0.sort_order',
            ],
        ];
    }

    #[DataProvider('failsDataProvider')]
    public function test_バリデーションエラーになること(array $data, string $expected_error_key): void
    {
        $validator = Validator::make($data, $this->rules());

        $this->assertTrue($validator->fails());
        $this->assertArrayHasKey($expected_error_key, $validator->errors()->toArray());
    }

    public function test_複数行のうち一部だけ不正な場合そのインデックスのみエラーになること(): void
    {
        $validator = Validator::make([
            'departments' => [
                ['id' => null, 'name' => '経理部', 'memo' => null, 'sort_order' => 0],
                ['id' => null, 'name' => '', 'memo' => null, 'sort_order' => 1],
            ],
        ], $this->rules());

        $errors = $validator->errors()->toArray();

        $this->assertTrue($validator->fails());
        $this->assertArrayNotHasKey('departments.0.name', $errors);
        $this->assertArrayHasKey('departments.1.name', $errors);
    }
}
